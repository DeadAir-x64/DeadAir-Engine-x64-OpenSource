# [DA_PORT] Прогонный стенд под санитайзеры. Разбор — в main.cpp и в da_port/tools/asan.
#
# ЗАЧЕМ ОТДЕЛЬНАЯ ЦЕЛЬ. Запустить под санитайзерами саму игру нельзя: выделенный сервер жёстко
# требует модуль рендера, R4 существует только под DX11, а GL мы из сборки убрали и он отстал на
# 25 ошибок. Заглушка IRender — это 124 виртуальных метода, большинство из которых обязаны
# возвращать пригодные объекты.
#
# Стенд обходит это целиком: он не рендерит и не грузит уровень, а дёргает подсистемы напрямую.
# Ловит ровно тот класс, что мы весь цикл вычёсывали руками, — обращение к освобождённому, выход за
# границы, переполнение. ⛔ Гонки уровня кадра он НЕ ловит: для них нужен настоящий запуск.
add_executable(da_asan_probe)

target_sources(da_asan_probe PRIVATE
    main.cpp
)

target_include_directories(da_asan_probe
    PRIVATE
    "${CMAKE_SOURCE_DIR}/src"
)

target_link_libraries(da_asan_probe
    PRIVATE
    xrCore
    xrAPI              # глобальный GEnv, нужен заголовкам xrCDB
    xrCDB              # пирамида отсечения и пространственная база
    xrMaterialSystem   # материалы: вылеты #37 и #47
)
